+/**
+ * Съёмка мокапов S2. Живёт только в одноразовом worktree.
+ *
+ *   node scripts/shoot-mockups.mjs --dist dist-a --variant a --out ../docs/design/mockups/demo-followups
+ *
+ * Числа считает браузер (getBoundingClientRect), не глаз. Пересечения считаются
+ * в координатах вьюпорта на конкретной позиции прокрутки — иначе fixed-элемент
+ * и содержимое страницы сравнивать нельзя.
+ *
+ * ── Почему съёмка разведена по проходам ──────────────────────────────────────
+ *
+ * Снимок БЛОКА требует спрятать плавающие слои: Playwright режет по рамке
+ * элемента, а липкая шапка ложится поверх. Первая редакция прятала их и
+ * возвращала обратно на той же странице — и восстановление молча не сработало
+ * после ВТОРОГО снимка блока: в поставку уехали снимки экрана подвала без
+ * кнопки чата, то есть ровно те кадры, ради которых съёмка и делалась.
+ * Причину в самом восстановлении воспроизвести не удалось (в изолированной
+ * пробе оно работает), поэтому чинится не причина, а класс: снимки блоков
+ * делаются на ОТДЕЛЬНОЙ загрузке страницы, после которой на ней больше ничего
+ * не снимается. Состояние, от которого зависят следующие кадры, не мутируется
+ * вовсе.
+ *
+ * И постусловие: каждый снимок экрана, который обязан содержать чат,
+ * проверяется по пикселю в центре кнопки. Без этого «кнопки нет» и «кнопка не

+ * проверить».

+const MEASURE = (selectors) => {
+  const rect = (el) => {
+    if (!el) return null;
+    const r = el.getBoundingClientRect();
+    if (r.width === 0 && r.height === 0) return null;
+    return { x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height) };
+  };
+  /** Рамка ТЕКСТА, а не блока: блок метки тянется во всю ширину контейнера, и
+      пересечение с ним говорит лишь о попадании в пустоту. */
+  const textRect = (el) => {
+    if (!el) return null;
+    const r = document.createRange();
+    r.selectNodeContents(el);
+    const b = r.getBoundingClientRect();
+    if (b.width === 0 && b.height === 0) return null;
+    return { x: Math.round(b.x), y: Math.round(b.y), w: Math.round(b.width), h: Math.round(b.height) };
+  };
+  const out = {};
+  for (const [name, sel] of Object.entries(selectors)) {
+    const el = document.querySelector(sel);
+    out[name] = name.endsWith('Text') ? textRect(el) : rect(el);
+  }
+  const overlap = (a, b) => {
+    if (!a || !b) return 0;
+    const w = Math.max(0, Math.min(a.x + a.w, b.x + b.w) - Math.max(a.x, b.x));
+    const h = Math.max(0, Math.min(a.y + a.h, b.y + b.h) - Math.max(a.y, b.y));
+    return w * h;
+  };
+  const gapX = (a, b) => (a && b ? Math.max(a.x, b.x) - Math.min(a.x + a.w, b.x + b.w) : null);
+  const chatBtn = out.chatButton;
+  const chatPanel = out.chatPanel;
+  const cover = {};
+  for (const [name, r] of Object.entries(out)) {
+    if (name === 'chatButton' || name === 'chatPanel' || !r) continue;
+    const area = r.w * r.h;
+    const byBtn = overlap(chatBtn, r);
+    const byPanel = overlap(chatPanel, r);
+    if (byBtn || byPanel) {
+      cover[name] = {
+        coveredByButtonPx: byBtn,
+        coveredByButtonPct: area ? Math.round((byBtn / area) * 100) : 0,
+        coveredByPanelPx: byPanel,
+        coveredByPanelPct: area ? Math.round((byPanel / area) * 100) : 0,
+      };
+    }
+  }
+  const gaps = {};
+  if (out.publishedText && chatBtn) gaps.publishedTextToChatButtonX = gapX(out.publishedText, chatBtn);
+  if (out.socialRow && out.publishedText) {
+    gaps.socialRowToPublishedTextY = Math.round(out.publishedText.y - (out.socialRow.y + out.socialRow.h));
+    gaps.socialRowOverlapsPublishedTextPx = overlap(out.socialRow, out.publishedText);
+  }
+  return {
+    rects: out, cover, gaps,
+    scrollY: Math.round(window.scrollY),
+    docHeight: Math.round(document.documentElement.scrollHeight),
+  };

+      const el = [...list.querySelectorAll('*')].find((n) => n.textContent.trim() === t);
+      if (!el) { found.push(`${t}: подписи нет`); continue; }
+      const b = el.getBoundingClientRect();
+      if (b.width === 0 || b.height === 0) { found.push(`${t}: рамка нулевая`); continue; }
+      // Рамки мало: снимок блока делается после того, как скрипт прячет
+      // плавающие слои через visibility, а visibility рамку не схлопывает.

+      // отсутствовал бы на картинке — это и есть «не смогла проверить» под
+      // видом «дефектов нет». Проверено мутацией: см. negative-check.txt.
+      const cs = getComputedStyle(el);
+      if (cs.visibility === 'hidden' || cs.display === 'none' || Number(cs.opacity) === 0) {
+        found.push(`${t}: узел невидим (visibility=${cs.visibility}, display=${cs.display}, opacity=${cs.opacity})`);
+        continue;
+      }
+      const inside = b.left >= sb.left - 1 && b.right <= sb.right + 1 && b.top >= sb.top - 1 && b.bottom <= sb.bottom + 1;
+      if (!inside) { found.push(`${t}: рамка вне секции`); continue; }
+      found.push(null);
+    }
+    const bad = found.filter(Boolean);
+    return bad.length ? { ok: false, why: bad.join('; ') } : { ok: true, count: titles.length };
+  }, AWARD_TITLES);
+  if (!r.ok) problems.push(`${path}: знаки наград не подтверждены — ${r.why}`);
+  else awardChecks.push(path);

+async function assertChatOnShot(page, path) {
+  if (VARIANT === 'main') return;
+  const btn = await page.evaluate(() => {
+    const el = document.querySelector('[data-mockup-chat-button]');
+    if (!el) return null;
+    const r = el.getBoundingClientRect();
+    return { x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height) };
+  });
+  if (!btn) {
+    problems.push(`${path}: кнопки чата нет в DOM`);
+    return;
+  }
+  const { data, info } = await sharp(path).raw().toBuffer({ resolveWithObject: true });
+  const x0 = Math.max(0, btn.x);
+  const y0 = Math.max(0, btn.y);
+  const x1 = Math.min(info.width, btn.x + btn.w);
+  const y1 = Math.min(info.height, btn.y + btn.h);
+  if (x1 <= x0 || y1 <= y0) {
+    problems.push(`${path}: рамка кнопки (${btn.x},${btn.y},${btn.w}x${btn.h}) вне кадра ${info.width}x${info.height}`);
+    return;
+  }
+  let hit = 0;
+  let total = 0;
+  for (let y = y0; y < y1; y++) {
+    for (let x = x0; x < x1; x++) {
+      const i = (y * info.width + x) * info.channels;
+      total++;
+      if (
+        Math.abs(data[i] - CHAT_RGB[0]) <= 10 &&
+        Math.abs(data[i + 1] - CHAT_RGB[1]) <= 10 &&
+        Math.abs(data[i + 2] - CHAT_RGB[2]) <= 10
+      ) hit++;
+    }
+  }
+  const pct = Math.round((hit / total) * 100);
+  // Вписанный круг даёт ~78 % рамки, белый значок съедает часть; на кадрах с
+  // раскрытой панелью её тень перекрывает край и доля падает до ~52 %. Порог 25 %
+  // отделяет «нарисована» от «нет вовсе» (0 %) с большим запасом в обе стороны.
+  if (pct < 25) {
+    problems.push(`${path}: в рамке кнопки ${pct}% пикселей цвета чата (порог 25%), кнопка не нарисована`);
+  }
+  chatChecks.push({ path, rect: { x: x0, y: y0, w: x1 - x0, h: y1 - y0 }, pct });
+}

+/** Доля пикселей цвета кнопки в заданной рамке КОНКРЕТНОГО файла. */
+async function chatFraction(file, r) {
+  const { data, info } = await sharp(file).raw().toBuffer({ resolveWithObject: true });
+  let hit = 0;
+  let total = 0;
+  for (let y = r.y; y < Math.min(info.height, r.y + r.h); y++) {
+    for (let x = r.x; x < Math.min(info.width, r.x + r.w); x++) {
+      const i = (y * info.width + x) * info.channels;
+      total++;
+      if (
+        Math.abs(data[i] - CHAT_RGB[0]) <= 10 &&
+        Math.abs(data[i + 1] - CHAT_RGB[1]) <= 10 &&
+        Math.abs(data[i + 2] - CHAT_RGB[2]) <= 10
+      ) hit++;
+    }
+  }
+  return total ? Math.round((hit / total) * 100) : 0;
+}
